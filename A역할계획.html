static std::queue<std::vector<uint8_t>> g_recvQueue; static CRITICAL_SECTION g_recvCS; void PushRecvPacket(std::vector<uint8_t> pkt) { EnterCriticalSection(&g_recvCS); g_recvQueue.push(std::move(pkt)); LeaveCriticalSection(&g_recvCS); } bool PopRecvPacket(std::vector<uint8_t>& out) { EnterCriticalSection(&g_recvCS); if (g_recvQueue.empty()) { LeaveCriticalSection(&g_recvCS); return false; } out = std::move(g_recvQueue.front()); g_recvQueue.pop(); LeaveCriticalSection(&g_recvCS); return true; }
